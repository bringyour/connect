// URNETWORK_DIAG_H1_GROUP_LINGER_US: when an H1-only sequence drains a ready group and no further Pack is queued, wait up
// to this long for one more Pack (per slot) so sparse upstream ACK frames share one websocket message.
var diagH1GroupLinger = func() time.Duration {
	v, _ := strconv.Atoi(os.Getenv("URNETWORK_DIAG_H1_GROUP_LINGER_US"))
	return time.Duration(v) * time.Microsecond
}()

var diagH1LingerHits, diagH1LingerMiss atomic.Uint64

func policyH1OnlyForDiag(p transferFlightPolicySnapshot) bool { return p.h1Only }
